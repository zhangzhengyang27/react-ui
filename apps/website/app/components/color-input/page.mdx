# ColorInput

ColorInput 是带弹出式 ColorPicker 的颜色输入组件，支持手动输入、色板选择和取色器。

## 基础使用

<Demo
    title="基础用法"
    description="点击输入框展开颜色选择器，支持直接输入 HEX 颜色值。"
    code={`import { ColorInput } from '@react-ui/ui'

export default function ColorInputBasicDemo() {
    return <ColorInput label="选择颜色" placeholder="选择颜色" defaultValue="#228be6" />
}`}
>
    <BasicDemo />
</Demo>

## 预设色板

<Demo
    title="预设色板"
    description="通过 swatches 属性提供一组常用颜色，方便快速选择。"
    code={`import { ColorInput } from '@react-ui/ui'

const SWATCHES = ['#25262b', '#868e96', '#fa5252', '#e64980', '#be4bdb', '#7950f2', '#4c6ef5']

export default function ColorInputSwatchesDemo() {
    return <ColorInput label="带预设色板" defaultValue="#228be6" swatches={SWATCHES} />
}`}
>
    <SwatchesDemo />
</Demo>

## 禁用取色器

<Demo
    title="无取色器"
    description="隐藏右侧取色器按钮，仅保留输入和色板选择。"
    code={`import { ColorInput } from '@react-ui/ui'

export default function ColorInputNoEyeDropperDemo() {
    return <ColorInput label="无取色器" defaultValue="#40c057" withEyeDropper={false} />
}`}
>
    <NoEyeDropperDemo />
</Demo>
