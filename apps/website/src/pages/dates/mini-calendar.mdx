## 无障碍

使用 `nextControlProps` 和 `previousControlProps` 为导航按钮添加 `aria-label` 和其他 props：

```tsx
import { MiniCalendar } from '@react-ui/ui';

function Demo() {
  return (
    <MiniCalendar
      nextControlProps={{ 'aria-label': 'Next range' }}
      previousControlProps={{ 'aria-label': 'Previous range' }}
    />
  );
}
```
