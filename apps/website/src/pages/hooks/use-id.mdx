import { Layout } from '@/layout';
import { MDX_DATA } from '@/mdx';

export default Layout(MDX_DATA.useId);

## 用法

`use-id` Hook 生成一个在多次渲染间保持不变的随机 id。
它通常用于将 input 元素与 label 绑定。
生成的随机 id 保存在 ref 中，除非组件卸载，否则不会改变。

```tsx
import { useId } from '@react-ui/hooks';

function Input({ id }: { id?: string }) {
  const uuid = useId(id);

  return (
    <>
      <label htmlFor={uuid}>输入标签</label>
      <input type="text" id={uuid} />
    </>
  );
}

// input 和 label 将拥有 id 'my-id'
const withId = <Input id="my-id" />;

// input 和 label 将拥有随机 id 'ui-fZMoF'
const withoutId = <Input />;
```

## 类型定义

```tsx
function useId(id: string): string;
```
