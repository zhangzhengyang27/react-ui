import { UseNetworkDemos } from '@react-ui/docs-demos';
import { Layout } from '@/layout';
import { MDX_DATA } from '@/mdx';

export default Layout(MDX_DATA.useNetwork);

## 用法

`use-network` Hook 返回一个包含当前连接状态的对象：

<Demo data={UseNetworkDemos.usage} />

## 浏览器支持

`use-network` 使用实验性的 `navigator.connection` API。请查看[浏览器兼容性表](https://developer.mozilla.org/en-US/docs/Web/API/Navigator/connection#browser_compatibility)。

## 类型定义

```tsx

interface UserNetworkReturnValue {
  online: boolean;
  downlink?: number;
  downlinkMax?: number;
  effectiveType?: 'slow-2g' | '2g' | '3g' | '4g';
  rtt?: number;
  saveData?: boolean;
  type?: 'bluetooth' | 'cellular' | 'ethernet' | 'wifi' | 'wimax' | 'none' | 'other' | 'unknown';
}

function useNetwork(): UserNetworkReturnValue;
```

## 导出类型

`UserNetworkReturnValue` 类型从 `@react-ui/hooks` 包导出，
可在应用中导入：

```tsx
import type { UserNetworkReturnValue } from '@react-ui/hooks';
```
